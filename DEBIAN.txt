CONFIG_HOSTNAME() {
  local NEW_HOSTNAME="$1"
  if [[ "$(hostnamectl --static)" -ne "$NEW_HOSTNAME" ]]
  then
    SUDO hostnamectl hostname "$NEW_HOSTNAME"
  fi
}

CONF_HOSTS_GENERATE() {
  CONF_HOSTS_CONTENT="127.0.0.1 localhost\n127.0.1.1 ${CONF_HOSTS_ENTRY}\n::1 localhost ip6-localhost ip6-loopback\nff02::1 ip6-allnodes\nff02::2 ip6-allrouters\n"
}

CONF_HOSTS_UPDATE() {
  if [[ $(grep -c -F "127.0.1.1" "$CONF_HOSTS_PATH") -eq 1 ]] && grep -q '^127\.0\.1\.1 .*$' "$CONF_HOSTS_PATH"
  then
    if ! grep -q "^127.0.1.1 $CONF_HOSTS_ENTRY$" "$CONF_HOSTS_PATH"
    then
      SUDO sed -i "s|^127\.0\.1\.1 .*$|127.0.1.1 ${CONF_HOSTS_ENTRY}|" "$CONF_HOSTS_PATH"
    fi
  fi
}

CONFIG_HOSTS() {
  local CFG_MODE="$1"
  local NEW_HOSTNAME="$2"
  local SITE_DOMAIN="$3"
  local CONF_HOSTS_ENTRY=""
  if [[ -z "$SITE_DOMAIN" ]]
  then
    CONF_HOSTS_ENTRY="$NEW_HOSTNAME"
  else
    CONF_HOSTS_ENTRY="${NEW_HOSTNAME}.${SITE_DOMAIN} ${NEW_HOSTNAME}"
  fi
  local CONF_HOSTS_PATH="/etc/hosts"
  CONF_READABLE "$CONF_HOSTS_PATH" || return 1
  SUDO CONF_WRITABLE "$CONF_HOSTS_PATH" || return 1
  if [[ "$CFG_MODE" eq "replace" ]]
  then
    CONF_HOSTS_CONTENT=""
    CONF_HOSTS_GENERATE
    printf "%b" "$CONF_HOSTS_CONTENT" | SUDO tee "$CONF_HOSTS_PATH" >/dev/null
  elif [[ "$CFG_MODE" eq "update" ]]
  then
    CONF_HOSTS_UPDATE
  fi
}

CONFIG_RESOLVE() {
  local CFG_MODE="$1"
  local SITE_DOMAIN="$2"
  local RESOLV_PATH="/etc/resolv.conf"
  if [[ "$CFG_MODE" == "default" ]]
  then
    CONF_WRITABLE "$RESOLV_PATH" || return 1
  else
    CONF_READABLE "$RESOLV_PATH" || return 1
  fi
}

DEFAULT_RESOLV() {
  local DNS_SRV_LIST=()
  while [[ $# -gt 0 ]]
  do
    case "$1" in
      --domain)
        SITE_DOMAIN="$2"
        shift 2
        ;;
      --dns-servers)
        shift
        while [[ $# -gt 0 && "${1:0:2}" != "--" ]]
        do
          DNS_SRV_LIST+=("$1")
          shift
        done
        ;;
    esac
  done
  CONF_WRITABLE "$RESOLV_PATH" || return 1
  local RESOLV_CONTENT=""
  if [[ -n "$SITE_DOMAIN" ]]
  then
    RESOLV_CONTENT+="search ${SITE_DOMAIN}\n"
  fi
  if [[ ${#DNS_SRV_LIST[@]} -gt 0 ]]
  then
    for DNS_SRV in "${DNS_SRV_LIST[@]}"
    do
      RESOLV_CONTENT+="nameserver ${DNS_SRV}\n"
    done
  fi
  printf "%b" "$RESOLV_CONTENT" | SUDO tee "$RESOLV_PATH" >/dev/null
}

UPDATE_RESOLV() {
  CONF_READABLE "$RESOLV_PATH" || return 1
  if [[ $(grep -c -F "search" "$RESOLV_PATH") -eq 1 ]] && grep -q '^search .*$' "$RESOLV_PATH"
  then
    if ! grep -q "^search $SITE_DOMAIN$" "$RESOLV_PATH"
    then
      CONF_WRITABLE "$RESOLV_PATH" || return 1
      SUDO sed -i "s|^search .*$|search ${SITE_DOMAIN}|" "$RESOLV_PATH"
    fi
  fi
}

#!/bin/bash

FILE_READABLE() {
  local FILE_PATH="$1"
  [[ -n "$FILE_PATH" && -f "$FILE_PATH" && -r "$FILE_PATH" ]] || {
    ERROR_MSG="${FILE_PATH} File READ Permission"
    return 1
  }
}

FILE_WRITABLE() {
  local FILE_PATH="$1"
  [[ -n "$FILE_PATH" && -f "$FILE_PATH" && -w "$FILE_PATH" ]] || {
    ERROR_MSG="${CONF_PATH} File WRITE Permission"
    return 1
  }
}

APP_EXECUTABLE() {
  local APP_PATH="$1"
  [[ -n "$APP_PATH" && -f "$APP_PATH" && -x "$APP_PATH" ]] || {
    ERROR_MSG="${APP_PATH} Application EXECUTE Permission"
    return 1
  }
}

APP_INSTALLED() {
  local APP_NAME="$1"
  [[ -n "$APP_NAME" ]] || {
    ERROR_MSG="Application Not Specified"
    return 1
  }
  command -v "$APP_NAME" >/dev/null 2>&1 || {
    ERROR_MSG="${APP_NAME} Application Not Installed"
    return 1
  }
}

PROMPT_SUDO() {
  if [[ $EUID -ne 0 ]]
  then
    APP_INSTALLED sudo || return 1
    sudo -v || {
      ERROR_MSG="SUDO Authentication Failed"
      return 1
    }
  fi
}

SUDO() {
  if [[ $EUID -eq 0 ]]
  then
    "$@"
  else
    sudo "$@"
  fi
}

APP_AVAILABLE() {
  local APP_NAME="$1"
  [[ -n "$APP_NAME" ]] || return 1
  if ! APP_INSTALLED "$APP_NAME" && ! APP_EXECUTABLE "./$APP_NAME"
  then
    ERROR_MSG="${APP_NAME} Application Not Available"
    return 1
  fi
}

PKG_INSTALLED() {
  local PKG_NAME="$1"
  [[ -n "$PKG_NAME" ]] || {
    ERROR_MSG="Package Not Specified"
    return 1
  }
  dpkg-query -W -f='${Status}' "$PKG_NAME" 2>/dev/null | grep -q "^install ok installed$" || {
    ERROR_MSG="${PKG_NAME} Package Not Installed"
    return 1
  }
}

UPDATE_PKGS_INDEX() {
  [[ "${PKGS_INDEX_UPDATED:-}" == "true" ]] || {
    PROMPT_SUDO || return 1
    export DEBIAN_FRONTEND="noninteractive"
    export NEEDRESTART_MODE="a"
    if SUDO apt-get update >/dev/null
    then
      PKGS_INDEX_UPDATED=true
    else
      ERROR_MSG="APT Packages Index Update Failed"
      return 1
    fi
  }
}

INSTALL_PKG() {
  local PKG_NAME="$1"
  [[ -n "$PKG_NAME" ]] || {
    ERROR_MSG="Package Not Specified"
    return 1
  }
  PKG_INSTALLED "$PKG_NAME" || {
    PROMPT_SUDO || return 1
    export DEBIAN_FRONTEND="noninteractive"
    export NEEDRESTART_MODE="a"
    SUDO apt-get install -y "$PKG_NAME" >/dev/null || {
      ERROR_MSG="${PKG_NAME} Package Installation Failed"
      return 1
    }
  }
}

UPGRADE_PKG() {
  local PKG_NAME="$1"
  [[ -n "$PKG_NAME" ]] || {
    ERROR_MSG="Package Not Specified"
    return 1
  }
  PROMPT_SUDO || return 1
  export DEBIAN_FRONTEND="noninteractive"
  export NEEDRESTART_MODE="a"
  SUDO apt-get install -y "$PKG_NAME" >/dev/null || {
    ERROR_MSG="${PKG_NAME} Package Upgrade Failed"
    return 1
  }
}

PROVIDE_APPS() {
  local -n APPS_LIST=$1
  local -n PKGS_LIST=$2
  local INDEX_UPDATED=false
  for APP_INDEX in "${!APPS_LIST[@]}"
  do
    local APP_NAME="${APPS_LIST[$APP_INDEX]}"
    if ! CHECK_APP "$APP_NAME"
    then
      local PKG_NAME="${PKGS_LIST[$APP_INDEX]}"
      if ! CHECK_PKG "$PKG_NAME"
      then
        if [[ "$INDEX_UPDATED" == "false" ]]
        then
          if ! UPDATE_PKG_INDEX
          then
            ERROR_MSG="Package Index Update Failed"
            return 1
          else
            INDEX_UPDATED=true
          fi
        fi
        if ! INSTALL_PKG "$PKG_NAME"
        then
          ERROR_MSG="$PKG_NAME Package Installation Failed"
          return 1
        fi
      fi
    fi
  done
}
