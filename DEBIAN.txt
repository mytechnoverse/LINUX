  fi

DEFAULT_HOSTS() {
  local HOSTS_CONTENT="127.0.0.1 localhost\n127.0.1.1 ${HOSTS_ENTRY}\n::1 localhost ip6-localhost ip6-loopback\nff02::1 ip6-allnodes\nff02::2 ip6-allrouters\n"
  printf "%b" "$HOSTS_CONTENT" | SUDO tee "$HOSTS_PATH" >/dev/null
}

UPDATE_HOSTS() {
  if [[ $(grep -c -F "127.0.1.1" "$HOSTS_PATH") -eq 1 ]] && grep -q '^127\.0\.1\.1 .*$' "$HOSTS_PATH"
  then
    if ! grep -q "^127.0.1.1 $HOSTS_ENTRY$" "$HOSTS_PATH"
    then
      CONF_WRITABLE "$HOSTS_PATH" || return 1
      SUDO sed -i "s|^127\.0\.1\.1 .*$|127.0.1.1 ${HOSTS_ENTRY}|" "$HOSTS_PATH"
    fi
  fi
}

CONFIG_HOSTS() {
  local CFG_MODE="$1"
  local NEW_HOSTNAME="$2"
  local SITE_DOMAIN="$3"
  local HOSTS_PATH="/etc/hosts"
  local HOSTS_ENTRY=""
  if [[ -z "$SITE_DOMAIN" ]]
  then
    HOSTS_ENTRY="$NEW_HOSTNAME"
  else
    HOSTS_ENTRY="${NEW_HOSTNAME}.${SITE_DOMAIN} ${NEW_HOSTNAME}"
  fi
  if [[ "$CFG_MODE" == "default" ]]
  then
    CONF_WRITABLE "$RESOLV_PATH" || return 1
    return DEFAULT_HOSTS
  else
    CONF_READABLE "$RESOLV_PATH" || return 1
    return UPDATE_HOSTS
  fi
}

CONFIG_RESOLVE() {
  local CFG_MODE="$1"
  local SITE_DOMAIN="$2"
  local RESOLV_PATH="/etc/resolv.conf"
  if [[ "$CFG_MODE" == "default" ]]
  then
    CONF_WRITABLE "$RESOLV_PATH" || return 1
  else
    CONF_READABLE "$RESOLV_PATH" || return 1
  fi
}

DEFAULT_RESOLV() {
  local DNS_SRV_LIST=()
  while [[ $# -gt 0 ]]
  do
    case "$1" in
      --domain)
        SITE_DOMAIN="$2"
        shift 2
        ;;
      --dns-servers)
        shift
        while [[ $# -gt 0 && "${1:0:2}" != "--" ]]
        do
          DNS_SRV_LIST+=("$1")
          shift
        done
        ;;
    esac
  done
  CONF_WRITABLE "$RESOLV_PATH" || return 1
  local RESOLV_CONTENT=""
  if [[ -n "$SITE_DOMAIN" ]]
  then
    RESOLV_CONTENT+="search ${SITE_DOMAIN}\n"
  fi
  if [[ ${#DNS_SRV_LIST[@]} -gt 0 ]]
  then
    for DNS_SRV in "${DNS_SRV_LIST[@]}"
    do
      RESOLV_CONTENT+="nameserver ${DNS_SRV}\n"
    done
  fi
  printf "%b" "$RESOLV_CONTENT" | SUDO tee "$RESOLV_PATH" >/dev/null
}

UPDATE_RESOLV() {
  CONF_READABLE "$RESOLV_PATH" || return 1
  if [[ $(grep -c -F "search" "$RESOLV_PATH") -eq 1 ]] && grep -q '^search .*$' "$RESOLV_PATH"
  then
    if ! grep -q "^search $SITE_DOMAIN$" "$RESOLV_PATH"
    then
      CONF_WRITABLE "$RESOLV_PATH" || return 1
      SUDO sed -i "s|^search .*$|search ${SITE_DOMAIN}|" "$RESOLV_PATH"
    fi
  fi
}

#!/bin/bash

CONF_READABLE() {
  local CONF_PATH="$1"
  [[ -n "$CONF_PATH" && -f "$CONF_PATH" && -r "$CONF_PATH" ]] || {
    ERROR_MSG="${CONF_PATH} Conf File Without READ Permission"
    return 1
  }
}

CONF_WRITABLE() {
  local CONF_PATH="$1"
  [[ -n "$CONF_PATH" && -f "$CONF_PATH" && -w "$CONF_PATH" ]] || {
    ERROR_MSG="${CONF_PATH} Conf File Without WRITE Permission"
    return 1
  }
}

APP_EXECUTABLE() {
  local APP_PATH="$1"
  [[ -n "$CONF_PATH" && -f "$CONF_PATH" && -x "$CONF_PATH" ]] || {
    ERROR_MSG="${APP_PATH} Application Without EXECUTE Permission"
    return 1
  }
}

SUDO() {
  if [[ $EUID -eq 0 ]]
  then
    "$@"
  else
    sudo "$@"
  fi
}

CHECK_APP() {
  local APP_NAME="$1"
  if ! command -v "$APP_NAME" >/dev/null 2>&1 && [[ ! -x "./$APP_NAME" ]]
  then
    return 1
  fi
}

CHECK_PKG() {
  local PKG_NAME="$1"
  dpkg-query -W -f='${Status}' "$PKG_NAME" 2>/dev/null | grep -Fq "install ok installed"
}

UPDATE_PKG_INDEX() {
  SUDO apt-get update >/dev/null
}

INSTALL_PKG() {
  local PKG_NAME="$1"
  SUDO apt-get install -y "$PKG_NAME" >/dev/null
}

PROVIDE_APPS() {
  local -n APPS_LIST=$1
  local -n PKGS_LIST=$2
  local INDEX_UPDATED=false
  export DEBIAN_FRONTEND=noninteractive
  for APP_INDEX in "${!APPS_LIST[@]}"
  do
    local APP_NAME="${APPS_LIST[$APP_INDEX]}"
    if ! CHECK_APP "$APP_NAME"
    then
      local PKG_NAME="${PKGS_LIST[$APP_INDEX]}"
      if ! CHECK_PKG "$PKG_NAME"
      then
        if [[ "$INDEX_UPDATED" == "false" ]]
        then
          if ! UPDATE_PKG_INDEX
          then
            ERROR_MSG="Package Index Update Failed"
            return 1
          else
            INDEX_UPDATED=true
          fi
        fi
        if ! INSTALL_PKG "$PKG_NAME"
        then
          ERROR_MSG="$PKG_NAME Package Installation Failed"
          return 1
        fi
      fi
    fi
  done
}
