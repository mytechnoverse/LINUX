# trap _bashutils_error_handler ERR

_bashutils_error_handler() {
  declare -i _bashutils_error_code=$?
  if [[ -n "$_bashutils_error_message" ]] && declare -p _bashutils_error_message &>/dev/null
  then
    printf '\n  MESSAGE : %s !' "$_bashutils_error_message" >&2
  fi
  (( _bashutils_error_code -eq 0 )) && return
  if [[ -v BASH_COMMAND && -n "$BASH_COMMAND" ]]
  then
    printf '\n  COMMAND : %s' "$BASH_COMMAND" >&2
  fi
  [[ ! -v FUNCNAME ]] && return "$_bashutils_error_code"
  declare -i _bashutils_functions_count
  _bashutils_functions_count=${#FUNCNAME[@]}
  (( $_bashutils_functions_count -lt 2 )) && return "$_bashutils_error_code"
  [[ ! -v BASH_SOURCE || ! -v BASH_LINENO ]] && return "$_bashutils_error_code"
  (( ${#BASH_SOURCE[@]} -lt _bashutils_functions_count || ${#BASH_LINENO[@]} -lt _bashutils_functions_count )) && return "$_bashutils_error_code"
  printf '\n  Call Stack Trace :' >&2
  declare -i bashutils_functions_index
  for (( bashutils_functions_index=${#FUNCNAME[@]}-1; bashutils_functions_index -ge 1; bashutils_functions_index-- ))
  do
    printf '\n  %s() [%s:%s]' "${FUNCNAME[bashutils_functions_index]}" "${BASH_SOURCE[bashutils_functions_index]}" "${BASH_LINENO[bashutils_functions_index-1]}" >&2
  done
  printf '\n' >&2
  return "$_bashutils_error_code"
}

# bashutils_init

_bashutils_init() {
  _bashutils_init_guard
  set -Eeuo pipefail
  declare -g _bashutils_error_message
  trap _bashutils_error_handler ERR
}

# _require_variable variable_name 

_require_variable() {
  declare variable_name="$1"
  [[ -n "$variable_name" ]] || {
    _bashutils_error_message="variable name required"
    return 1
  }
  [[ -v variable_name ]] || {
    _bashutils_error_message="${variable_name} variable required"
    return 1
  }
}

# _require_parameter parameter_name || return 1 

_require_parameter() {
  declare parameter_name="$1"
  [[ -n "$parameter_name" ]] || {
    bashutils_error_message="${parameter_name} missing"
    return 1
  }
  declare -n parameter_refrence="$parameter_name"
  [[ -n "$parameter_refrence" ]] || {
    bashutils_error_message="${parameter_name} parameter required"
    return 1
  }
}

#!/bin/bash

bashutils_init

data_options=()
bool_options=()

bashutils_options=$(getopt --longoptions help-banner:,bool-options:,data-options:,required-options:,script-version:,version,help -- "$@")
eval set -- "$bashutils_options"

while true
do
  case "$1" in
    --help-banner)
      help_banner="$2"
      shift 2 ;;
    --bool-options)
      shift
      while [[ $# -gt 0 && ${1:0:2} != "--" ]]
      do
        bool_options+=("$1")
        shift
      done ;;
    ---data-options)
      shift
      while [[ $# -gt 0 && ${1:0:2} != "--" ]]
      do
        data_options+=("$1")
        shift
      done ;;
    --required-options)
      shift
      while [[ $# -gt 0 && ${1:0:2} != "--" ]]
      do
        required_options+=("$1")
        shift
      done ;;
    --script-version)
      script_version="$2"
      shift 2 ;;
    --version)
      printf '%s' "$version"
      exit 0 ;;
    --help)
      printf '  %s %s\n' "$0" "$help_banner"
      exit 0 ;;
    --)
      shift
      break ;;
    *)
      bashutils_error_message="$0 SCRIPT INVALID OPTION $1"
      exit 1 ;;
  esac

#!/bin/bash

help_banner=$script_help_banner

  $script_bool_variable_name=true
  EOF)
done

do
  script_options+="${data_option}:,"
  script_data_variable_name="${data_option//-/_}"
  script_template+=$(cat <<EOF
  $script_data_variable_name= ?????
  EOF)
done

script_options="${script_options},version,help"

script_options=$script_options

#!/bin/bash
set -Eeuo pipefail

usage() {
  printf ' %s --addr 192.168.1.2/24 --gateway 192.168.1.1\n' "$0"
}

INT_IP_CIDR=""
INT_GATEWAY=""

ARGS=$(getopt --long addr:,gateway:,help -- "$@")
eval set -- "$ARGS"

while true; do
  case "$1" in
    --addr)
      INT_IP_CIDR="$2"
      shift 2 ;;
    --gateway)
      INT_GATEWAY="$2"
      shift 2 ;;
    --help)
      usage
      exit 1 ;;
    --)
      shift
      break ;;
    *)
      usage
      exit 1 ;;
  esac
done

INT_MAC=$( ip -j link | jq -r '[ .[] | select(.link_type == "ether") | .address ][0]')

if [[ -z "$INT_MAC" ]]
then
  printf 'Ethernet Network Interface Not Found !\n' >&2
  exit 1
fi

YQQ_QUERY='.Match.Name = "eth0" | .Network.DHCP = "no" | .Network.IPv6AcceptRA = "no"'

if [[ -n "$INT_IP_CIDR" ]]
then
  YQQ_QUERY+=' | .Network.Address = strenv(INT_IP_CIDR)'
else
  usage
  exit 1
fi

if [[ -n "$INT_GATEWAY" ]]
then
  YQQ_QUERY+=' | .Route[0].Destination = "0.0.0.0/0" | .Route[0].Gateway = strenv(INT_GATEWAY)'
else
  usage
  exit 1
fi

INT_CONF_PREFIX="/etc/systemd/network/10-eth0."
INT_CONF_L2_PATH="${INT_CONF_PREFIX}link"
INT_CONF_L3_PATH="${INT_CONF_PREFIX}network"
if systemctl is-active networking --quiet
then
  rm -rf /etc/systemd/network/*
  touch "$INT_CONF_L2_PATH"
  chown root:root "$INT_CONF_L2_PATH"
  chmod 644 "$INT_CONF_L2_PATH"
  yqq -n -o=ini '.Match.MACAddress = strenv(INT_MAC) | .Link.Name = "eth0"' > "$INT_CONF_L2_PATH"
  touch "$INT_CONF_L3_PATH"
  chown root:root "$INT_CONF_L3_PATH"
  chmod 644 "$INT_CONF_L3_PATH"
  yqq -n -o=ini "$YQQ_QUERY" > "$INT_CONF_L3_PATH"
  systemctl enable systemd-networkd >/dev/null
  systemctl mask networking --quiet
  reboot
else
  yqq -i -p=ini -o=ini "$YQQ_QUERY" "$INT_CONF_L3_PATH"
  systemctl restart systemd-networkd
fi

CONF_HOSTS_UPDATE() {
  if [[ $(grep -c -F "127.0.1.1" "$CONF_HOSTS_PATH") -eq 1 ]] && grep -q '^127\.0\.1\.1 .*$' "$CONF_HOSTS_PATH"
  then
    if ! grep -q "^127.0.1.1 $CONF_HOSTS_ENTRY$" "$CONF_HOSTS_PATH"
    then
      SUDO sed -i "s|^127\.0\.1\.1 .*$|127.0.1.1 ${CONF_HOSTS_ENTRY}|" "$CONF_HOSTS_PATH"
    fi
  fi
}


ADD_STATIC_DNS_RECORD() {
  local DOMAIN_NAME="$1"
  local A_RECORD="$2"
  local HOSTS_FILE_PATH="/etc/hosts"
  local ACTIVE_LINES=$(grep -v -E '^[[:space:]]*#.*$' "$HOSTS_FILE_PATH")
  local TARGET_LINES=$(printf '%s\n' "$ACTIVE_LINES" | grep -E '[[:space:]]'"$DOMAIN_NAME"'([[:space:]]|$)')
  if [[ -n "$TARGET_LINES" ]]
  then
    local TARGET_LINES_COUNT=$(printf '%s\n' "$TARGET_LINES" | wc -l)
    [[ $TARGET_LINES_COUNT -eq 1 ]] || {
      ERROR_MSG="${DOMAIN_NAME} DUPLICATE Entries"
      return 1
    }
    local DOMAIN_REGEX=${DOMAIN_NAME//./\\.}
    if ! printf '%s\n' "$TARGET_LINES" | grep -q -E "^[[:space:]]*${A_RECORD//./\\.}[[:space:]].*[[:space:]]'"$DOMAIN_REGEX"'([[:space:]]|$)"
    then
      ERROR_MSG="${DOMAIN_NAME} Entry SYNTAX"
      return 1
    fi
  else
    printf '%s\n' "${A_RECORD} ${DOMAIN_NAME}" | sudo tee -a "$HOSTS_FILE_PATH" > /dev/null
  fi
}

CONFIG_HOSTS() {
  local CFG_MODE="$1"
  local NEW_HOSTNAME="$2"
  local SITE_DOMAIN="$3"
  local CONF_HOSTS_ENTRY=""
  if [[ -z "$SITE_DOMAIN" ]]
  then
    CONF_HOSTS_ENTRY="$NEW_HOSTNAME"
  else
    CONF_HOSTS_ENTRY="${NEW_HOSTNAME}.${SITE_DOMAIN} ${NEW_HOSTNAME}"
  fi
  local CONF_HOSTS_PATH="/etc/hosts"
  CONF_READABLE "$CONF_HOSTS_PATH" || return 1
  SUDO CONF_WRITABLE "$CONF_HOSTS_PATH" || return 1
  if [[ "$CFG_MODE" eq "replace" ]]
  then
    CONF_HOSTS_CONTENT=""
    CONF_HOSTS_GENERATE
    printf "%b" "$CONF_HOSTS_CONTENT" | SUDO tee "$CONF_HOSTS_PATH" >/dev/null
  elif [[ "$CFG_MODE" eq "update" ]]
  then
    CONF_HOSTS_UPDATE
  fi
}

DEFAULT_RESOLV() {
  local DNS_SRV_LIST=()
  while [[ $# -gt 0 ]]
  do
    case "$1" in
      --domain)
        SITE_DOMAIN="$2"
        shift 2
        ;;
      --dns-servers)
        shift
        while [[ $# -gt 0 && "${1:0:2}" != "--" ]]
        do
          DNS_SRV_LIST+=("$1")
          shift
        done
        ;;
    esac
  done
  CONF_WRITABLE "$RESOLV_PATH" || return 1
  local RESOLV_CONTENT=""
  if [[ -n "$SITE_DOMAIN" ]]
  then
    RESOLV_CONTENT+="search ${SITE_DOMAIN}\n"
  fi
  if [[ ${#DNS_SRV_LIST[@]} -gt 0 ]]
  then
    for DNS_SRV in "${DNS_SRV_LIST[@]}"
    do
      RESOLV_CONTENT+="nameserver ${DNS_SRV}\n"
    done
  fi
  printf "%b" "$RESOLV_CONTENT" | SUDO tee "$RESOLV_PATH" >/dev/null
}

UPDATE_RESOLV() {
  CONF_READABLE "$RESOLV_PATH" || return 1
  if [[ $(grep -c -F "search" "$RESOLV_PATH") -eq 1 ]] && grep -q '^search .*$' "$RESOLV_PATH"
  then
    if ! grep -q "^search $SITE_DOMAIN$" "$RESOLV_PATH"
    then
      CONF_WRITABLE "$RESOLV_PATH" || return 1
      SUDO sed -i "s|^search .*$|search ${SITE_DOMAIN}|" "$RESOLV_PATH"
    fi
  fi
}

PROVIDE_APPS() {
  local -n APPS_LIST=$1
  local -n PKGS_LIST=$2
  local INDEX_UPDATED=false
  for APP_INDEX in "${!APPS_LIST[@]}"
  do
    local APP_NAME="${APPS_LIST[$APP_INDEX]}"
    if ! CHECK_APP "$APP_NAME"
    then
      local PKG_NAME="${PKGS_LIST[$APP_INDEX]}"
      if ! CHECK_PKG "$PKG_NAME"
      then
        if [[ "$INDEX_UPDATED" == "false" ]]
        then
          if ! UPDATE_PKG_INDEX
          then
            ERROR_MSG="Package Index Update Failed"
            return 1
          else
            INDEX_UPDATED=true
          fi
        fi
        if ! INSTALL_PKG "$PKG_NAME"
        then
          ERROR_MSG="$PKG_NAME Package Installation Failed"
          return 1
        fi
      fi
    fi
  done
}
