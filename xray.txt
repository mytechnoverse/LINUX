====== TLS Setup :

openssl genpkey -algorithm EC -pkeyopt ec_paramgen_curve:P-256 -out ./internal-ca.key
certificate_serial=$(openssl rand -hex 16)
openssl req -new -sha256 -x509 -days 3650 -key ./internal-ca.key -out ./internal-ca.crt -subj "/CN=internal-ca" -set_serial "0x${certificate_serial}" \
-addext "basicConstraints=critical,CA:TRUE" -addext "keyUsage=critical,keyCertSign,cRLSign" -addext "subjectKeyIdentifier=hash"

openssl genpkey -algorithm EC -pkeyopt ec_paramgen_curve:P-256 -out ./xyz.internal.key
openssl req -new -sha256 -key ./xyz.internal.key -out xyz.internal.csr -subj "/CN=xyz.internal"
certificate_serial=$(openssl rand -hex 16)

cert_extensions=$(cat <<EOF
basicConstraints=critical,CA:FALSE
keyUsage=critical,digitalSignature
subjectKeyIdentifier=hash
authorityKeyIdentifier=keyid,issuer
extendedKeyUsage=serverAuth
subjectAltName=DNS:xyz.internal
EOF
)

openssl x509 -req -sha256 -days 3650 -CA ./internal-ca.crt -CAkey ./internal-ca.key -in ./xyz.internal.csr -out ./xyz.internal.crt -set_serial "0x${certificate_serial}" -extfile <(printf '%s' "$cert_extensions")
rm ./xyz.internal.csr

openssl verify -CAfile ./internal-ca.crt ./xyz.internal.crt
cat xyz.internal.crt internal-ca.crt > xyz.internal-fullchain.crt
