ECDSA ( secp256r1 | P-256 | prime256v1 ) with SHA256 for wazuh

-- checking certificate against a CA :

openssl verify -CAfile ca.crt server.crt

--- checking certificate against a private key :

openssl x509 -noout -in server.crt -pubkey
openssl pkey -noout -in server.key -pubout -passin file:<(printf '%s\n' "$password")

--- checking certificate detail :

openssl x509 -text -noout -in certificate.pem 

--- checking request detail :

openssl req -text -noout -in request.pem

--- checking private key detail :

openssl pkey -text -noout -in private.key

--- indexer ( extendedKeyUsage ) extensions :

openssl ecparam -genkey -name prime256v1 -out indexer.key
openssl req -new -key indexer.key -out indexer.csr \
  -subj "/CN=wazuh-indexer" \
  -addext "subjectAltName=DNS:wazuh-indexer,DNS:localhost,IP:127.0.0.1" \
  -addext "extendedKeyUsage=serverAuth,clientAuth"

--- private key passing :

openssl pkey -in <(printf '%s\n' "$private_key")

declare first_line
declare file_path
get_first_line() {
  declare -n first_line_ref=$1
  declare file_path=$2
  read -r first_line_ref < $file_path
}

is_file_encoding_pem() {
  [[ "$first_line" =~ ^-----BEGIN[[:space:]].*-----$ ]]
}

is_file_certificate() {
  [[ "$first_line" =~ ^-----BEGIN[[:space:]]CERTIFICATE-----$ ]] && openssl x509 -noout -in $file_path
}

is_file_cerificate_request() {
  [[ "$first_line" =~ ^-----BEGIN[[:space:]]CERTIFICATE[[:space:]]REQUEST-----$ ]] && openssl req -noout -in $file_path
}

is_file_format_pkcs8() {
  [[ "$first_line" =~ ^-----BEGIN[[:space:]](ENCRYPTED[[:space:]]+)?PRIVATE[[:space:]]KEY-----$ ]]
}

is_private_key_encrypted() {
  [[ "$first_line" =~ ^-----BEGIN[[:space:]]ENCRYPTED[[:space:]]PRIVATE[[:space:]]KEY-----$ ]]
}

is_file_private_key() {
  is_file_format_pkcs8
  if is_private_key_encrypted $file_path
  then
    openssl pkey -noout -in $file_path -check -passin file:<(printf '%s\n' "$password") || return 1
  else
    openssl pkey -noout -in $file_path || return 1
  fi
}

--- Generating Private Key : ( PKCS8 , No Private Key Encyption , No Authentication )

openssl genpkey -algorithm EC -pkeyopt ec_paramgen_curve:P-256 -out ./root-ca.key
chmod 600 ./root-ca.key

--- Generating Encrypted Private Key : ( PKCS8 , Private Key Encyption , No Authentication )

V1 : -v1 PBE-SHA1-3DES
V2 : -v2 aes-256-cbc -v2prf hmacWithSHA256 -saltlen 16 -iter 1000000

openssl pkcs8 -topk8 -v2 aes-256-cbc -v2prf hmacWithSHA256 -saltlen 16 -iter 1000000 -in <(openssl genpkey -algorithm EC -pkeyopt ec_paramgen_curve:P-256) -out ./root-ca.key -passout file:<(printf '%s\n' "$password")
chmod 444 ./root-ca.key

--- Generating PFX Bundle : ( PKCS12 , No Private Key Encyption , Bundle Authentication )

pfx_key_password='xyz'
openssl pkcs12 -export -in root-ca.pem -out root-ca.pfx -macalg SHA256 -macsaltlen 16 -certpbe NONE -iter 1000000 -passout file:<(printf '%s' "$pfx_key_password")

--- Generating Encrypted PFX Bundle : ( PKCS12 , Private Key Encyption , Bundle Authentication )

private_key_password='xyz'
pfx_key_password='xyz'
