chmod 0644 $apt_repo_public_key_path
read -r -d $'\x1c' apt_repo_sources <<-EOF
	Types: deb
	URIs: $apt_repo_url
	Suites: $apt_repo_suite
	Components: $apt_repo_component
	Signed-By: $apt_repo_public_key_path
$(printf '\x1c')
EOF

chmod 0644 $apt_repo_sources_path

apt_repo_packages_file_url="${apt_repo_url}/dists/${apt_repo_suite}/${apt_repo_component}/binary-amd64/Packages"
apt_repo_packages_file=$(curl --fail --silent --show-error --location --compressed --connect-timeout 3 "$apt_repo_packages_file_url")

declare -A packages_dictionary
apt_repo_preferences_packages=''

while read -r -d $'\n' this_line
do
	[[ $this_line =~ ^Package:[[:space:]](.*)$ ]] && {
		package_name=${BASH_REMATCH[1]}
		[[ -v packages_dictionary[$package_name] ]] || {
			apt_repo_preferences_packages+="${package_name} "
			packages_dictionary[$package_name]=''
		}
	}
done < <(printf '%s\n' "$apt_repo_packages_file")

read -r -d $'\x1c' apt_repo_preferences <<-EOF
	Package: $apt_repo_preferences_packages
	Pin: release o=$apt_repo_name
	Pin-Priority: 900
$(printf '\x1c')
EOF

printf '%s\n' "$apt_repo_preferences" > $apt_repo_preferences_path

declare first_line
declare file_path
get_first_line() {
  declare -n first_line_ref=$1
  declare file_path=$2
  read -r first_line_ref < $file_path
}

is_file_encoding_pem() {
  [[ "$first_line" =~ ^-----BEGIN[[:space:]].*-----$ ]]
}

is_file_certificate() {
  [[ "$first_line" =~ ^-----BEGIN[[:space:]]CERTIFICATE-----$ ]] && openssl x509 -noout -in $file_path
}

is_file_cerificate_request() {
  [[ "$first_line" =~ ^-----BEGIN[[:space:]]CERTIFICATE[[:space:]]REQUEST-----$ ]] && openssl req -noout -in $file_path
}

is_file_format_pkcs8() {
  [[ "$first_line" =~ ^-----BEGIN[[:space:]](ENCRYPTED[[:space:]]+)?PRIVATE[[:space:]]KEY-----$ ]]
}

is_private_key_encrypted() {
  [[ "$first_line" =~ ^-----BEGIN[[:space:]]ENCRYPTED[[:space:]]PRIVATE[[:space:]]KEY-----$ ]]
}

is_file_private_key() {
  is_file_format_pkcs8
  if is_private_key_encrypted $file_path
  then
    openssl pkey -noout -in $file_path -check -passin file:<(printf '%s\n' "$password") || return 1
  else
    openssl pkey -noout -in $file_path || return 1
  fi
}
