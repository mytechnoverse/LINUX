#!/bin/bash

command -v gpg >/dev/null 2>&1 || exit 1

[[ -x ./wazuh-certs-tool.sh ]] || chmod +x ./wazuh-certs-tool.sh

rm -rf ./wazuh-certificates


read -s -r -p "Enter Password : " GPG_PASS
echo

wazuh_component_name="indexer"
wazuh_component_cert_cn="wazuh-indexer"
wazuh_component_cert_san="wazuh-indexer.internal"
wazuh_server_type="master"

yq_query='
  .nodes[strenv(wazuh_component_name)][0].name = strenv(wazuh_component_cert_cn) |
  .nodes[strenv(wazuh_component_name)][0].ip = strenv(wazuh_component_cert_san)
'

[[ -z "$wazuh_server_type" ]] || {
  yq_query+=' |
    .nodes.server[0].node_type = strenv(wazuh_server_type)
  '
}

wazuh_component_name="$wazuh_component_name" \
wazuh_component_cert_cn="$wazuh_component_cert_cn" \
wazuh_component_cert_san="$wazuh_component_cert_san" \
wazuh_server_type="$wazuh_server_type" \
